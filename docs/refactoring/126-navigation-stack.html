// App/CHALLAApp/Sources/MainFeature.swift  — 새로
@Reducer
struct MainFeature {
    @Reducer
    enum Path {                              // 스택에 쌓일 수 있는 화면 전부
        case roomDetail(RoomDetailFeature)
        case photoDetail(PhotoDetailFeature)
        case chat(ChatRoomFeature)
        case roomSettings(RoomSettingsFeature)
        case roomCoverEdit(RoomCoverEditFeature)
        case setting(SettingFeature)
        case profileEdit(ProfileSetupFeature)
    }
    @ObservableState
    struct State: Equatable {
        @Shared(.appProfile) var profile: UserProfile
        var home: HomeFeature.State
        var path = StackState<Path.State>()
        @Presents var camera: LiveCameraFeature.State?
    }
    enum Action {
        case home(HomeFeature.Action)
        case path(StackActionOf<Path>)
        case camera(PresentationAction<LiveCameraFeature.Action>)
        case pushNotificationTapped(roomID: Room.ID)          // 딥링크 (지금 TODO)
    }
    var body: some ReducerOf<Self> {
        Scope(state: \.home, action: \.home) { HomeFeature() }
        Reduce { state, action in
            switch action {
            // 지금 AppFeature.swift:314–321 — "교체" 대신 "쌓기"
            case let .home(.delegate(.roomSelected(card))):
                state.path.append(.roomDetail(RoomDetailFeature.State(room: card.room)))
                return .none
            // 지금 :392–402
            case let .path(.element(id: _, action: .roomDetail(.delegate(.photoTapped(photoID))))):
                state.path.append(.photoDetail(…))
                return .none
            // 지금 :331–341 — 카메라는 스택 밖
            case let .home(.delegate(.cameraRequested(entry))):
                state.camera = LiveCameraFeature.State(…)
                return .none
            // 딥링크 — 두 화면을 한 번에 쌓으면 끝
            case let .pushNotificationTapped(roomID):
                state.path.removeAll()
                state.path.append(.roomDetail(…)); state.path.append(.photoDetail(…))
                return .none
            default: return .none
            }
        }
        .forEach(\.path, action: \.path)
        .ifLet(\.$camera, action: \.camera) { LiveCameraFeature() }
    }
}

회의에서 정할 것 세 가지

결정 1 · 설정 스택

설정은 자기 NavigationStack(SettingView.swift:20)으로 테마·알림·계정을 push한다. 앱 스택에 합칠지(SettingFeature 수정, 스와이프 일관) fullScreenCover로 따로 띄울지(무수정).
추천: 합치기. 하위 화면이 3개뿐이다.

결정 2 · 돌아올 때 재조회

지금은 새로 만들어서 자동 재조회된다. 스택에선 화면이 살아 있으니 필요한 곳만 명시적으로: 촬영 후엔 바뀐 방만 갱신, 방 상세 복귀 시 사진·리액션은 onAppear 조회 또는 delegate 반영.

결정 3 · 커스텀 전환 유지

ScreenTransition은 UIKit push/pop을 재현한 것. 시스템 NavigationStack·fullScreenCover가 같은 모양을 주므로 삭제가 기본안. 디자인이 똑같은지는 실기기 비교.

스와이프 복구: 채팅·사진 상세는 시스템 바를 숨겨서(ChatRoomView.swift:32) 스와이프가 꺼지는데, 되살리는 코드가 이미 있다(SettingFeature/…/InteractivePopGestureEnabler.swift). 공용 모듈로 올려 앱 스택 루트에 한 번 적용.

DI는 손대지 않는다

UseCase 주입은 TCA 정석 그대로다. 다른 점 하나는 진짜 구현(live)을 UseCase 파일이 아니라 앱 시작 지점 CompositionRoot에서 꽂는 것인데, 이건 모듈을 나눈 프로젝트의 정석 변형이다.

  • 정의·등록·사용 전부 정석. @DependencyClient → TestDependencyKey → DependencyValues → @Dependency → .run에서 호출. CreateRoomUseCase.swift:8–53 · CreateRoomFeature.swift:61, 120–130
  • live는 한 곳에서. CHALLAApp.swift:31 prepareDependencies → CompositionRoot.swift:190. Domain 모듈이 서버 코드를 모르게 하려는 설계.
  • 스택으로 바꿔도 변경 없음. 스택에 쌓인 자식 화면은 의존성을 자동으로 물려받고, dismiss는 TCA가 원소마다 넣어 준다. 데모 앱도 그대로 돈다.
  • 테스트도 두 층 다 정석. Feature는 TestStore + withDependencies, UseCase 규칙은 .live(가짜 Repository)로 TCA 없이. CreateRoomFeatureTests.swift:10–23 · CreateRoomUseCaseLiveTests.swift
  • 주의 하나: live가 자동이 아니라서 새 UseCase를 CompositionRoot에 안 꽂으면 실행 중 "구현 안 됨"이 불린다. PR 체크리스트에.

    case photoDetail(PhotoDetailFeature)

예전엔 "지금 화면 하나"만 저장해서, 방 상세로 가면 홈이 지워졌다. 지금은 홈과 목록이 따로 있어서 방 상세로 가도 홈이 남는다.

② 열기 = 배열에 추가, 닫기 = 배열에서 빼기

// 홈에서 방을 고름 → 방 상세를 위에 추가
case let .home(.delegate(.roomSelected(card))):
    state.path.append(.roomDetail(RoomDetailFeature.State(room: card.room)))

// 방 상세에서 뒤로 버튼 → 그 화면만 뺌
case let .path(.element(id: id, action: .roomDetail(.delegate(.closeTapped)))):
    state.path.pop(from: id)

화면들은 서로를 모른다. "열어 줘 / 닫아 줘"를 delegate로 알리기만 하고, 실제로 추가·삭제는 MainFeature가 한다. .path(.element(id:, action:))는 "목록의 id번 화면이 보낸 액션"이라는 뜻이다. 스와이프 뒤로가기는 코드가 없다 — 시스템이 알아서 같은 결과를 낸다.

③ 화면 — 앱에 NavigationStack이 하나

// MainView.swift

    HomeView(store: store.scope(state: \.home, action: \.home))

path 배열이 늘면 화면이 밀려 들어오고, 줄면 빠진다. 전환 애니메이션과 스와이프는 iOS 기본 것을 쓴다. 화면 하나를 추가하려면 ①의 Path에 한 줄, 여기에 한 줄이면 된다.

⑤ 커버 수정을 스와이프로 닫으면 대신 저장

단계 3 — 카메라 (#127 · 상진)

카메라를 AppFeature의 enum 화면에서 MainFeature의 @Presents로 옮겼다. 이제 카메라는 메인을 통째로 갈아끼우지 않고 홈·방 상세 위를 fullScreenCover로 덮었다가 걷힌다. 아래 화면이 살아 있으니 돌아갈 곳(CameraOrigin)을 기억할 필요가 없어졌다.

① 열기 — 스택 밖에 덮는다

// MainFeature.State
@Presents var camera: LiveCameraFeature.State?

// 홈 · 방 상세 어느 쪽에서 와도 같다
case let .home(.delegate(.cameraRequested(entry))):
    guard state.camera == nil else { return .none }
    state.camera = LiveCameraFeature.State(entry: entry)

// body
.ifLet(\.$camera, action: \.camera) { LiveCameraFeature() }

스택(path)은 건드리지 않는다. 진입 버튼이 방·필터·권한을 이미 갖춰 둔 요청이라 받는 즉시 띄운다.

② 화면 — MainView가 덮는다

.fullScreenCover(item: $store.scope(state: \.camera, action: \.camera)) { liveStore in
    let cameraStore = liveStore.scope(state: \.camera, action: \.camera)
    CameraView(store: cameraStore) {
        LiveCameraPreview(session: cameraSession, store: cameraStore)
    }
    .overlay(alignment: .top) { cameraOverlay }   // 참여 토스트
}

프리뷰용 @Dependency(\.cameraSession)도 AppView에서 MainView로 옮겼다. fullScreenCover는 뷰 계층 위에 따로 떠서 RootView의 토스트 overlay를 가리므로, 같은 토스트를 RootView → AppView → MainView로 넘겨 커버 안에도 얹는다.

③ 닫기 · 촬영 완료

case .camera(.presented(.camera(.delegate(.closeRequested)))):
    state.camera = nil
    return refreshScreenBehindCamera(state)      // 드러난 홈 또는 방 상세에 view(.task)

case let .camera(.presented(.camera(.delegate(.captureFinished(roomID))))):
    state.camera = nil
    return showShotRoom(id: roomID, &state)
    // 맨 위가 찍은 방 상세 → 그대로 두고 highlightsNewestPhoto 켜고 재조회
    // 아니면 → 홈 카드에서 방을 찾아 그 방 상세 한 장으로 스택 교체

예전엔 메인을 새로 만들어서 저절로 재조회됐다. fullScreenCover는 아래 화면을 가린 적이 없어 그 뷰의 .task가 다시 돌지 않으므로 직접 보낸다. 다른 방을 찍었을 때 상세를 쌓지 않고 교체하는 것은 "스택의 방 상세는 한 장뿐"이라는 전제(roomDetail(roomID:))를 지키기 위해서다.

알아둘 동작

  • 촬영 중에는 참여 토스트로 방을 열지 않는다. 판별이 screenID != .camera에서 MainFeature의 camera == nil 가드로 옮겨 갔다. 토스트 자체는 카메라 위에도 뜬다.
  • 초대 링크는 카메라를 닫는다. 홈에서 입장을 이어야 하는데 카메라가 덮고 있으면 홈이 보이지 않는다. MainFeature 안에서 닫아야 ifLet이 같은 액션 안에서 카메라 이펙트를 취소한다 (밖에서 닫으면 다음 MainFeature 액션 때 취소된다).
